# Reference mesh input file for create_mesh.bash
#
# Lists EVERY parameter. Your own input file only needs the keys that your
# refinement and cut use (see each section); other keys are ignored.
# Format: key=value, one per line; '#' starts a comment. Lengths in km,
# angles in degrees. Run:  bash create_mesh.bash [your_input_file.txt]
#
# A mesh is a REFINEMENT (where cells are fine) combined with a CUT (whether
# and how the globe is cut to a regional domain). All 9 combinations work:
#
#               cut=global        cut=circle         cut=ellipse
#  uniform      uniform globe     uniform circle     uniform ellipse
#  circle       circle on globe   circular domain    circle in elliptical domain
#  ellipse      ellipse on globe  ellipse in circle  elliptical domain
#=============================================================================#

#-----------------------------------------------------------------------------
# Location (always required)
#   lat, lon : center of the refinement and of the cut (degrees)
#-----------------------------------------------------------------------------
lat=-22.66697851
lon=-45.01576941

#-----------------------------------------------------------------------------
# Refinement (always required)
#   refinement = uniform | circle | ellipse
#   high_res   : finest cell size (km). Always required.
#   low_res    : cell size outside the refinement (km). circle and ellipse only;
#                must be larger than high_res.
#-----------------------------------------------------------------------------
refinement=circle
high_res=1.5
low_res=5

#-----------------------------------------------------------------------------
# Circle refinement parameters (refinement=circle only)
#   Cell size is high_res out to inner_radius, then grows linearly to low_res
#   at outer_radius, and is low_res beyond.
#   inner_radius : radius of the high-resolution circle (km), >= 0
#   outer_radius : radius where the transition ends (km), > inner_radius
#-----------------------------------------------------------------------------
inner_radius=300
outer_radius=400

#-----------------------------------------------------------------------------
# Ellipse refinement parameters (refinement=ellipse only)
#   Cell size is high_res inside the ellipse, grows linearly to low_res up to
#   the ellipse scaled by (1 + delta), and is low_res beyond.
#   a_km  : semi-axis along the rotated x direction (km)
#   b_km  : semi-axis along the rotated y direction (km)
#   angle : rotation of the ellipse, degrees counter-clockwise from east (default 0)
#   delta : transition width as a fraction of the ellipse size (default 0.12)
#-----------------------------------------------------------------------------
#a_km=2500
#b_km=3688
#angle=-45
#delta=1

#-----------------------------------------------------------------------------
# Cut (always required)
#   cut = global   no cut: a global mesh
#       = circle   regional domain inside a circle around (lat, lon)
#       = ellipse  regional domain inside an ellipse around (lat, lon)
#   The cut must lie where the cell size is constant (outside the refinement's
#   transition), otherwise the run stops and tells you the minimum valid size.
#-----------------------------------------------------------------------------
cut=circle

#-----------------------------------------------------------------------------
# Circle cut parameters (cut=circle only)
